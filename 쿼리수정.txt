-- 함수 추가                 
CREATE OR REPLACE FUNCTION FN_DETAIL_CODEID(AS_parentCode VARCHAR(20)) 
RETURNS VARCHAR(30) 
LANGUAGE plpgsql
AS $$
DECLARE 
    RTN_Code VARCHAR(30);
BEGIN 
    SELECT COALESCE(
               MAX(
                   CAST(
                       REPLACE(code, AS_parentCode || '_', '') AS INTEGER
                   )
               ), 0
           ) + 1 
    INTO RTN_Code 
    FROM COMTCCMMNDETAILCODE 
    WHERE CODE_ID = AS_parentCode;

    -- 접두사(AS_parentCode + '_')와 번호를 결합하여 최종 반환 값 완성
    RTN_Code := AS_parentCode || '_' || RTN_Code;

    RETURN RTN_Code;
END;
$$;


    
                
CREATE OR REPLACE FUNCTION  FN_ROLE_CNT(AS_ROLE_ID VARCHAR(20))
RETURNS INTEGER
LANGUAGE plpgsql
AS $$
DECLARE
    CNT INTEGER;
BEGIN
    SELECT COALESCE(COUNT(MENU_NO), 0)
    INTO CNT
    FROM COMTNMENUCREATDTLS
    WHERE ROLE_ID = AS_ROLE_ID;
    
    RETURN CNT;
END;
$$;

-- DROP FUNCTION public.fn_centerid();
CREATE OR REPLACE FUNCTION public.fn_centerid()
 RETURNS character
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_centerid CHAR(9);
BEGIN
    SELECT
        CASE
            WHEN (COALESCE(COUNT(*), 0) + 1) > 9  AND (COALESCE(COUNT(*), 0) + 1) < 100
            THEN 'C' || SUBSTRING(TO_CHAR(NOW(), 'YYYYMMDD'), 3, 8) || (COALESCE(COUNT(*), 0) + 1)::text
            WHEN (COALESCE(COUNT(*), 0) + 1) >= 100
            THEN 'C' || SUBSTRING(TO_CHAR(NOW(), 'YYYYMMDD'), 4, 8) || (COALESCE(COUNT(*), 0) + 1)::text
            ELSE 'C' || SUBSTRING(TO_CHAR(NOW(), 'YYYYMMDD'), 3, 8) || '0' || (COALESCE(COUNT(*), 0) + 1)::text
        END INTO v_centerid
    FROM tb_centerinfo
    WHERE TO_CHAR(center_regdate, 'YYYYMMDD') = TO_CHAR(NOW(), 'YYYYMMDD');

    RETURN v_centerid;
END;
$function$
;
